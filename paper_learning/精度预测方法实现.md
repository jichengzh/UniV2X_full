**1A.4 硬约束 DSL — 三类分别实现**:

```python
# framework/constraints.py
from dataclasses import dataclass
from typing import Callable

@dataclass
class Constraint:
    name: str
    kind: str  # 'physical' | 'empirical' | 'soft'
    check: Callable[[Config, HardwareCapability], tuple[bool, str]]

# 物理硬约束(违反 build 失败)
PHYSICAL_CONSTRAINTS = [
    Constraint(
        "trt_gpu_symmetric_only", "physical",
        lambda cfg, hw: (
            all(cfg.q_object[m] != "asymmetric" for m in cfg.q_object),
            "TRT GPU 强制对称量化"
        )
    ),
    Constraint(
        "dla_op_whitelist", "physical",
        lambda cfg, hw: check_dla_whitelist(cfg, hw)
    ),
    Constraint(
        "dla_per_tensor_only", "physical",
        lambda cfg, hw: (
            all(cfg.d_routing[m] != "DLA" or cfg.q_granularity[m] == "per-tensor"
                for m in cfg.q_granularity),
            "DLA 路由模块强制 per-tensor"
        )
    ),
    # ...
]

# 经验硬约束(违反能跑但性能差)
EMPIRICAL_CONSTRAINTS = [
    Constraint(
        "channel_align_32", "empirical",
        lambda cfg, hw: check_channel_alignment(cfg, hw.alignment["int8_channel"])
    ),
    Constraint(
        "k_dim_ge_64", "empirical",
        lambda cfg, hw: check_k_dim(cfg, threshold=64)
    ),
    Constraint(
        "cross_layer_gradient_lt_30pct", "empirical",
        lambda cfg, hw: check_prune_gradient(cfg, max_jump=0.30)
    ),
]

# 软约束(违反不致命,作为惩罚项)
SOFT_CONSTRAINTS = [
    Constraint(
        "module_precision_consistency", "soft",
        lambda cfg, hw: check_precision_consistency(cfg)
    ),
]

def is_legal(config, hw, kinds=("physical", "empirical")):
    """硬约束过滤(soft 不在这里检查,留给搜索器作 penalty)"""
    all_constraints = PHYSICAL_CONSTRAINTS + EMPIRICAL_CONSTRAINTS
    for c in all_constraints:
        if c.kind not in kinds:
            continue
        ok, reason = c.check(config, hw)
        if not ok:
            return False, f"[{c.kind}/{c.name}] {reason}"
    return True, ""
```

**1A.5 双向传播逻辑**:

```python
# framework/propagation.py

def propagate_d_to_b2(config: Config, hw: HardwareCapability) -> Config:
    """D 路由 → B2 强制约束传播"""
    new_config = deepcopy(config)
    for module, route in config.d_routing.items():
        if route.startswith("DLA"):
            new_config.q_bits[module] = "INT8"
            new_config.q_granularity[module] = "per-tensor"
            new_config.q_object[module] = "W+A"
            # 输入布局强制 HWC4(在 build 时由 TRT 处理,这里只标记)
    return new_config

def propagate_b2_to_d(config: Config, hw: HardwareCapability) -> Config:
    """B2 选择 → D 路由可行集合收窄"""
    new_config = deepcopy(config)
    for module in config.q_granularity:
        if config.q_granularity[module] == "per-channel":
            # 该模块禁止 DLA
            if config.d_routing.get(module, "GPU").startswith("DLA"):
                new_config.d_routing[module] = "GPU"
    return new_config

def propagate_all(config: Config, hw: HardwareCapability) -> Config:
    """完整双向传播 — 迭代到不动点"""
    prev = None
    curr = config
    while curr != prev:
        prev = curr
        curr = propagate_d_to_b2(curr, hw)
        curr = propagate_b2_to_d(curr, hw)
    return curr
```

**1A.6 最小搜索器 v0**:

```python
# framework/searcher_v0.py

def random_search(hw: HardwareCapability, n_candidates=50, max_attempts=5000):
    """随机采样 + 硬约束过滤,无评估器"""
    candidates = []
    attempts = 0
    while len(candidates) < n_candidates and attempts < max_attempts:
        config = sample_random_config(hw)         # 均匀采样
        config = propagate_all(config, hw)        # 双向传播
        ok, _ = is_legal(config, hw)
        if ok:
            candidates.append(config)
        attempts += 1
    return candidates
```

import lightgbm as lgb

def fit_latency_xfer(common_configs_4090, common_configs_orin):
    """
    特征:
      - latency_4090 (主信号)
      - flops (compute intensity)
      - mem_intensity = bytes_moved / flops (memory-bound 标记)
      - has_dla (是否含 DLA 路由)
      - module_type one-hot (conv / matmul / attention)
    target:
      - latency_orin

    X = build_features(common_configs_4090)
    y = common_configs_orin["latency_ms"].values
    model = lgb.LGBMRegressor(num_leaves=15, n_estimators=100)
    model.fit(X, y)
    return model

# 验收: cross-validation R² > 0.85

### 4.3 关键设计

**2.1 模块级 LUT 维度展开**:

```
模块列表(UniV2X):
  - backbone (ResNet50 风格 CNN)
  - bev_encoder (Lift-Splat)
  - bev_decoder (Transformer-style)
  - detection_head (FCOS-style)
  - tracking_head (...)

每个模块的配置维度(笛卡儿积):
  - prune_rate ∈ {0, 0.1, 0.2, 0.3, 0.5, 0.7}                  6
  - q_bits ∈ {INT8, FP16}                                       2
  - q_granularity ∈ {per-tensor, per-channel}                   2
  - sparse_tc ∈ {off, 2:4}                                      2

每个模块组合数: 6 × 2 × 2 × 2 = 48
模块数 ~5 → 总组合 ~240
单次 trtexec(build + 100 iter): ~1.5 min
4090 上估计: 240 × 1.5min ≈ 6 h(单线程)/ 2-3 h(并行)
```

**2.1 实施流程**:

```python
# scripts/build_lut_4090.py

import itertools
from pathlib import Path

modules = ["backbone", "bev_encoder", "bev_decoder", "head"]
prune_rates = [0, 0.1, 0.2, 0.3, 0.5, 0.7]
q_bits = ["INT8", "FP16"]
q_gran = ["per-tensor", "per-channel"]
sparse = [False, True]

results = []
for mod in modules:
    for pr, qb, qg, sp in itertools.product(prune_rates, q_bits, q_gran, sparse):
        # 1. 生成对应 ONNX 子图
        onnx_path = generate_module_onnx(mod, prune_rate=pr)
        # 2. trtexec 实测(走 4090 GPU only)
        latency_p50, latency_p95 = run_trtexec(
            onnx_path, q_bits=qb, q_gran=qg, sparse=sp,
            warmup=10, iterations=100
        )
        results.append({
            "module": mod, "prune_rate": pr, "q_bits": qb,
            "q_granularity": qg, "sparse_tc": sp,
            "latency_p50_ms": latency_p50, "latency_p95_ms": latency_p95
        })

pd.DataFrame(results).to_parquet("data/latency_lut_4090.parquet")
```

**2.2 Orin Top-K 抽样策略**:

```
不可能在 Orin 上跑全部 240 组合(代价太高)
策略: 分层抽样
  - 每个模块至少 10 个组合(覆盖各 prune_rate 段)
  - 每段 prune_rate 至少各 q_bits 都跑一个
  - 含 DLA 路由的模块单独跑(因为 4090 上没数据)
共 ~50-80 组合,Orin 上 1.5-2 天

用途:
  1. 校验 4090 LUT(用 1B.3 的映射函数预测,对比实测)
  2. 拟合 4090 → Orin 残差校正函数
  3. DLA 路由 LUT(4090 没有)
```

**2.3 拐点验证 v2(对照 1B.6 扩展)**:

**S4.4 硬约束三种处理对比**:

| 方式 | 优劣 | 推荐 |

**Repair 示例**:
```python
class HardConstraintRepair(Repair):
    def _do(self, problem, X, **kwargs):
        for i in range(len(X)):
            # 通道对齐 32
            for module in modules:
                rate = X[i, f"prune_rate__{module}"]
                orig_ch = original_channels[module]
                pruned_ch = round_to_multiple(orig_ch * (1 - rate), 32)
                X[i, f"prune_rate__{module}"] = 1 - pruned_ch / orig_ch
            # DLA 路由触发 B2 强制
            for m in modules:
                if X[i, f"d_routing__{m}"] == "DLA":
                    X[i, f"q_bits__{m}"] = "INT8"
                    X[i, f"q_granularity__{m}"] = "per-tensor"
        return X
```

## 10. 风险与备选

| 阶段 | 风险 | 概率 | 影响 | 备选 |
|------|------|------|------|------|
| P1A | DLA 算子白名单不全 | 高 | 1A.4 漏过滤 | 用 trtexec 实测建动态白名单 |
| P1B | Orin 资源短缺 | 高 | Phase 1B 推迟 | 降级到 5 个 baseline,推迟非关键项到 S1 |
| P1B | 4090→Orin R² < 0.7 | 中 | LUT 跨平台失效 | S1 全部 LUT 都在 Orin 上建 |
| P1B | DLA 不能 build UniV2X | 中 | D 维度退化 | 锁死 D=GPU only(等价 Orin Nano 场景) |
| S1 | LUT 跑表中断 | 中 | 进度延误 | 加 checkpoint,每 50 个 flush |
| S2 | 数据 < 200 | 中 | S3 训不出来 | 降低主动采样阈值 / 用合成数据 |
| S3 | Spearman < 0.85 | 中 | 搜索器不可信 | 升级 MLP / 加派生特征 / ensemble |
| S4 | NSGA-II 收敛慢 | 低 | 1000 次评估不够 | 加大 budget / 改 NSGA-III |

| S5 | 4090→Orin 复现率 < 0.7 | 中 | Phase 3 不复现 | Phase 3 整体重做 / 强化 latency 映射 |
